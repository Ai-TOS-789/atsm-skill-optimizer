    embed_rank [label="Embed Rank
(Semantic Similarity)", fillcolor="#2196F3", shape=box, fontcolor=white, style="filled,rounded"];
    causal [label="Causal Inference

    goal_plan [label="Goal Planning
(Decomposition)", fillcolor="#2196F3", shape=box, fontcolor=white, style="filled,rounded"];
    skill_chain [label="Skill Chaining
(Sequence)", fillcolor="#2196F3", shape=box, fontcolor=white, style="filled,rounded"];
    execution [label="Execution
(Hermes CLI)", fillcolor="#FF9800", shape=box, fontcolor=white, style="filled,rounded"];
    verification [label="Verification
(Check Result)", fillcolor="#9C27B0", shape=box, fontcolor=white, style="filled,rounded"];

    // --- Edges ---
    input -> atsm_rank [label="task text"];
    input -> embed_rank [label="task text"];
    input -> causal [label="historical data"];
    atsm_rank -> goal_plan [label="ranked skills"];
    embed_rank -> goal_plan [label="semantic scores"];
    causal -> goal_plan [label="causal links"];
    goal_plan -> skill_chain [label="subgoals"];
    skill_chain -> execution [label="skill sequence"];
    execution -> verification [label="result"];
    verification -> atsm_rank [label="feedback (success/fail)"];
    verification -> embed_rank [label="feedback (embedding update)"];

    // --- Feedback Loop ---
    feedback [label="Feedback
Loop", shape=plaintext, fontcolor="#666666", fontsize=10];
    feedback -> verification [style=invis];
